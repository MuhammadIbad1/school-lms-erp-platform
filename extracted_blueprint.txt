Complete Cross-Module Cascade Workflows
1. Admission $\rightarrow$ Academic Assignment $\rightarrow$ Automatic Invoicing $\rightarrow$ Auxiliary Allocation
Step 1: Student profile is created under a specific Class and Section.
Step 2: System looks up the active Fee Master rules for that class.
Step 3: System automatically generates the student's initial Fee Invoice (Tuition + Admission Fee).
Step 4: If assigned a Transport Route or Hostel Room, corresponding monthly fee structures are appended to the invoice automatically.
2. Daily Attendance $\rightarrow$ Communication $\rightarrow$ Parent Portal
Step 1: Teacher opens the /teacher/attendance view for their assigned section.
Step 2: One-click batch selection marks student status (Present, Absent, Late, Excused).
Step 3: Saving the ledger fires an event that queues instant notifications to the linked Parent Profile and logs the data into the student's aggregate attendance percentage ledger.
3. Examinations $\rightarrow$ Grade Engine $\rightarrow$ Report Cards $\rightarrow$ Student/Parent Portals
Step 1: Admin sets up an Exam Term with subject-wise max and pass marks.
Step 2: Subject Teachers enter marks in the /teacher/marks matrix grid.
Step 3: Grade engine evaluates score ranges against Grade Rules (e.g., $\ge 90\% = \text{A+}$, $\ge 80\% = \text{A}$) and calculates cumulative GPA.
Step 4: Admin publishes results, unlocking printable PDF Report Cards on both Student and Parent Dashboards.
4. Teacher HR $\rightarrow$ Timetable Allocation $\rightarrow$ LMS Classroom Access
Step 1: Teacher profile is linked to specific Subjects and Sections via teacher_subject.
Step 2: Timetable engine assigns the teacher to specific time slots in the weekly schedule grid.
Step 3: LMS grants the teacher authorization to post study materials, assignments, and announcements exclusively for those assigned sections.

CREATE TABLE users (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(255) NOT NULL,
    email VARCHAR(255) UNIQUE NOT NULL,
    password VARCHAR(255) NOT NULL,
    phone VARCHAR(50) NULL,
    avatar VARCHAR(255) NULL,
    status ENUM('active', 'inactive', 'suspended') DEFAULT 'active',
    created_at TIMESTAMP NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
);
CREATE TABLE roles (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    guard_name VARCHAR(50) DEFAULT 'web'
);
CREATE TABLE model_has_roles (
    role_id BIGINT NOT NULL,
    model_type VARCHAR(255) NOT NULL,
    model_id BIGINT NOT NULL,
    PRIMARY KEY (role_id, model_type, model_id),
    FOREIGN KEY (role_id) REFERENCES roles(id) ON DELETE CASCADE
);
CREATE TABLE parent_profiles (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    user_id BIGINT NOT NULL UNIQUE,
    occupation VARCHAR(100) NULL,
    identity_card_no VARCHAR(100) NULL,
    address TEXT NULL,
    created_at TIMESTAMP NULL, updated_at TIMESTAMP NULL,
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE
);
CREATE TABLE teacher_profiles (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    user_id BIGINT NOT NULL UNIQUE,
    employee_code VARCHAR(50) NOT NULL UNIQUE,
    qualification VARCHAR(255) NULL,
    joining_date DATE NOT NULL,
    basic_salary DECIMAL(10,2) NOT NULL DEFAULT 0.00,
    created_at TIMESTAMP NULL, updated_at TIMESTAMP NULL,
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE
);

-- SUBSYSTEM 2: ACADEMIC HIERARCHY & ENROLLMENT

CREATE TABLE academic_years (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(50) NOT NULL, -- e.g. "2026-2027"
    start_date DATE NOT NULL,
    end_date DATE NOT NULL,
    is_current BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP NULL, updated_at TIMESTAMP NULL
);
CREATE TABLE classes (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(50) NOT NULL, -- e.g. "Grade 10"
    numeric_code INT NOT NULL,
    created_at TIMESTAMP NULL, updated_at TIMESTAMP NULL
);
CREATE TABLE sections (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    class_id BIGINT NOT NULL,
    name VARCHAR(50) NOT NULL, -- e.g. "Section A"
    capacity INT DEFAULT 40,
    created_at TIMESTAMP NULL, updated_at TIMESTAMP NULL,
    FOREIGN KEY (class_id) REFERENCES classes(id) ON DELETE CASCADE
);
CREATE TABLE subjects (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    class_id BIGINT NOT NULL,
    name VARCHAR(100) NOT NULL,
    code VARCHAR(20) NOT NULL,
    pass_marks INT DEFAULT 40,
    max_marks INT DEFAULT 100,
    type ENUM('theory', 'practical', 'both') DEFAULT 'theory',
    created_at TIMESTAMP NULL, updated_at TIMESTAMP NULL,
    FOREIGN KEY (class_id) REFERENCES classes(id) ON DELETE CASCADE
);
CREATE TABLE student_profiles (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    user_id BIGINT NOT NULL UNIQUE,
    parent_id BIGINT NULL,
    class_id BIGINT NOT NULL,
    section_id BIGINT NOT NULL,
    roll_number VARCHAR(50) NOT NULL,
    admission_number VARCHAR(50) NOT NULL UNIQUE,
    admission_date DATE NOT NULL,
    blood_group VARCHAR(10) NULL,
    created_at TIMESTAMP NULL, updated_at TIMESTAMP NULL,
    FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE,
    FOREIGN KEY (parent_id) REFERENCES parent_profiles(id) ON DELETE SET NULL,
    FOREIGN KEY (class_id) REFERENCES classes(id),
    FOREIGN KEY (section_id) REFERENCES sections(id)
);
CREATE TABLE teacher_subject (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    teacher_id BIGINT NOT NULL,
    subject_id BIGINT NOT NULL,
    section_id BIGINT NOT NULL,
    FOREIGN KEY (teacher_id) REFERENCES teacher_profiles(id) ON DELETE CASCADE,
    FOREIGN KEY (subject_id) REFERENCES subjects(id) ON DELETE CASCADE,
    FOREIGN KEY (section_id) REFERENCES sections(id) ON DELETE CASCADE
);

CREATE TABLE exams (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    academic_year_id BIGINT NOT NULL,
    name VARCHAR(100) NOT NULL, -- e.g. "Mid-Term Exam"
    start_date DATE NOT NULL,
    end_date DATE NOT NULL,
    is_published BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP NULL, updated_at TIMESTAMP NULL,
    FOREIGN KEY (academic_year_id) REFERENCES academic_years(id) ON DELETE CASCADE
);
CREATE TABLE exam_schedules (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    exam_id BIGINT NOT NULL,
    subject_id BIGINT NOT NULL,
    section_id BIGINT NOT NULL,
    exam_date DATE NOT NULL,
    start_time TIME NOT NULL,
    end_time TIME NOT NULL,
    room_number VARCHAR(50) NULL,
    FOREIGN KEY (exam_id) REFERENCES exams(id) ON DELETE CASCADE,
    FOREIGN KEY (subject_id) REFERENCES subjects(id) ON DELETE CASCADE,
    FOREIGN KEY (section_id) REFERENCES sections(id) ON DELETE CASCADE

CREATE TABLE marks (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    exam_id BIGINT NOT NULL,
    subject_id BIGINT NOT NULL,

    created_at TIMESTAMP NULL, updated_at TIMESTAMP NULL,
    UNIQUE KEY unique_student_exam_mark (exam_id, subject_id, student_id),
    FOREIGN KEY (exam_id) REFERENCES exams(id) ON DELETE CASCADE,
    FOREIGN KEY (subject_id) REFERENCES subjects(id) ON DELETE CASCADE,
    FOREIGN KEY (student_id) REFERENCES student_profiles(id) ON DELETE CASCADE,
    FOREIGN KEY (entered_by) REFERENCES users(id)
);
CREATE TABLE grade_rules (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    grade_name VARCHAR(10) NOT NULL, -- e.g. "A+", "A", "B"
    min_percentage DECIMAL(5,2) NOT NULL,
    max_percentage DECIMAL(5,2) NOT NULL,
    grade_point DECIMAL(3,2) NOT NULL,
    remarks VARCHAR(100) NULL
);

    FOREIGN KEY (subject_id) REFERENCES subjects(id) ON DELETE CASCADE,

    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    class_id BIGINT NOT NULL,

CREATE TABLE fee_invoices (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    student_id BIGINT NOT NULL,
    invoice_number VARCHAR(100) NOT NULL UNIQUE,
    title VARCHAR(255) NOT NULL,
    total_amount DECIMAL(10,2) NOT NULL,
    paid_amount DECIMAL(10,2) NOT NULL DEFAULT 0.00,
    due_date DATE NOT NULL,
    status ENUM('unpaid', 'partially_paid', 'paid') DEFAULT 'unpaid',
    created_at TIMESTAMP NULL, updated_at TIMESTAMP NULL,
    FOREIGN KEY (student_id) REFERENCES student_profiles(id) ON DELETE CASCADE
);
CREATE TABLE fee_payments (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    fee_invoice_id BIGINT NOT NULL,
    transaction_id VARCHAR(100) NOT NULL UNIQUE,
    amount_paid DECIMAL(10,2) NOT NULL,
    payment_method ENUM('cash', 'card', 'bank_transfer', 'online') NOT NULL,
    paid_at DATETIME NOT NULL,
    received_by BIGINT NOT NULL,
    FOREIGN KEY (fee_invoice_id) REFERENCES fee_invoices(id) ON DELETE CASCADE,
    FOREIGN KEY (received_by) REFERENCES users(id)
);
CREATE TABLE payrolls (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    teacher_id BIGINT NOT NULL,
    month_year VARCHAR(20) NOT NULL, -- e.g. "August 2026"
    basic_salary DECIMAL(10,2) NOT NULL,
    allowances DECIMAL(10,2) DEFAULT 0.00,
    deductions DECIMAL(10,2) DEFAULT 0.00,
    net_salary DECIMAL(10,2) NOT NULL,
    status ENUM('generated', 'paid') DEFAULT 'generated',
    paid_at DATE NULL,
    created_at TIMESTAMP NULL, updated_at TIMESTAMP NULL,
    FOREIGN KEY (teacher_id) REFERENCES teacher_profiles(id) ON DELETE CASCADE
);
